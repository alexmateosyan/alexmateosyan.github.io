Code sample: C hash table

/* From register_user(): update an existing user, or insert a new one */
int bucketIndex = get_bucket_for_user(user_id);
if (bucketIndex == -1) return ERROR;            // invalid user id

User *current = ht->buckets[bucketIndex];
while (current != NULL) {                       // walk the chain
    if (current->user_id == user_id) {
        strcpy(current->password, password);    // existing user: update
        return SUCCESS;
    }
    current = current->next;
}

User *newUser = (User *) malloc(sizeof(User));
if (newUser == NULL) return ERROR;             // allocation failed
newUser->user_id = user_id;
strcpy(newUser->password, password);
newUser->next = ht->buckets[bucketIndex];        // insert at head of list
ht->buckets[bucketIndex] = newUser;
return SUCCESS;
